```js
/* 首帧前把布局跑到收敛，这样用户看到的是「已经摆好」的图，不会看见乱跳 */
function kSettle(g, W, H, iters){
  let e = 1;
  for (let i = 0; i < iters && e > 0.05; i++) e = kStep(g, W, H);
  return e;
}

/* 一次性把视口对准整张图（只是移动镜头，不改变布局，也不限制图的大小） */
function kFit(g, W, H){
  if (!g.nodes.length) return;
  let a = Infinity, b = Infinity, c = -Infinity, d = -Infinity;
  g.nodes.forEach(nd => {
    a = Math.min(a, nd.x - nd.r); c = Math.max(c, nd.x + nd.r);
    b = Math.min(b, nd.y - nd.r); d = Math.max(d, nd.y + nd.r);
  });
  const gw = Math.max(c - a, 1), gh = Math.max(d - b, 1);
  const pad = 26;
  const s = Math.min((W - pad*2) / gw, (H - pad*2) / gh);
  g.scale = Math.max(0.25, Math.min(1.6, s));
  g.ox = W/2 - ((a + c)/2) * g.scale;
  g.oy = H/2 - ((b + d)/2) * g.scale;
}
```

- [ ] **Step 2: 实现分组数据与列表渲染**

在 `kLegend()` 之后插入：

```js
/* ---------- 📋 列表视图：一级（核心概念）/ 二级（扩展知识） ---------- */
function kListData(){
  const known = {};
  KNOW.forEach(n => { known[n.id] = n; });
  const cores = KNOW.filter(n => (n.layer || 0) === 0);
  const listed = {};
  const groups = cores.map(c => {
    listed[c.id] = true;
    const kids = [];
    (c.related || []).forEach(r => {
      const k = r && known[r.id];
      if (k && k.id !== c.id && !kids.some(x => x.id === k.id)){   // 跳过悬空与自引用
        kids.push(k);
        listed[k.id] = true;
      }
    });
    return { core: c, kids: kids };
  });
  const orphans = KNOW.filter(n => !listed[n.id]);   // 未被任何一级引用的扩展节点
  return { groups: groups, orphans: orphans };
}

function kListItem(n, sub){
  const ver = n.confidence === 'verified';
  const done = state.checks['know:'+n.id];
  return '<div class="kli'+(sub?' sub':'')+'" data-kopen2="'+esc(n.id)+'">'+
    '<span class="kln">'+esc(n.term || n.id)+'</span>'+
    (ver ? '<span class="kbadge ver">有来源</span>' : '<span class="kbadge ai">待查证</span>')+
    (done ? '<span class="kbadge done">已掌握</span>' : '')+
    '<span class="kla">打开卡片 ›</span>'+
  '</div>';
}

function kListView(){
  const d = kListData();
  let h = '<div class="klmeta">共 '+KNOW.length+' 个知识点：一级 '+d.groups.length+
          ' 个核心概念，二级 '+d.groups.reduce((a,g)=>a+g.kids.length,0)+' 个扩展知识</div>';
  h += '<div class="klist">';
  d.groups.forEach(g => {
    h += '<div class="klgrp">'+
      '<div class="klgh">一级 · 核心概念</div>'+
      kListItem(g.core, false)+
      (g.kids.length ? '<div class="klkids">'+
          '<div class="klgh sub">二级 · 扩展知识</div>'+
          g.kids.map(k => kListItem(k, true)).join('')+
        '</div>' : '')+
    '</div>';
  });
  if (d.orphans.length){
    h += '<div class="klgrp"><div class="klgh">未归类的知识点</div>'+
         d.orphans.map(n => kListItem(n, false)).join('')+'</div>';
  }
  return h + '</div>';
}
```
