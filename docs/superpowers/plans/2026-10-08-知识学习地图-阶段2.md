# 知识学习地图 · 阶段 2 实现计划（关系总览：力导向知识图谱）

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 在报告的「知识学习地图」模块里加一个「🗺 关系总览」切换视图：用 canvas 手绘力导向图谱展示知识点之间的关系网，可拖拽/缩放/平移，点节点能跳回对应知识卡片。

**Architecture:** 纯前端、零依赖、离线可用。沿用报告模板已有的 `KNOW`（知识节点数组）与 `#knowBox` 容器模式：新增一个 `kView` 视图开关，图谱态时把 `#knowBox` 渲染成 canvas 容器，并用一个自写的力模拟（斥力 + 弹簧 + 向心 + 阻尼）布局。模拟收敛后**停止 rAF 循环**省 CPU，拖拽时再唤醒。

**Tech Stack:** 原生 JS + Canvas 2D（无任何第三方库）。改造 `web/report_template.html` 单文件。

**参考：** 设计文档 §4.3；已验证可行的示意稿 `_mockups/B-力导向知识图谱.html`

**数据源：** `REPORT_DATA.knowledge` 的每个节点含 `{id, term, layer, category, confidence, related:[{id, relation}], ...}`。
- 节点 = 知识点；边 = `related`（**必须跳过悬空引用**：`related` 里可能指向不存在的 id）
- 颜色按 `layer`（0=核心蓝 / 1=扩展灰）；描边按 `confidence`（verified=绿 / ai-generated=琥珀）

- [ ] **Step 3: 新增力模拟与绘制**（插在 `bindKnow()` 之前）

```js
/* ---------- 🗺 力导向图谱（零依赖 canvas） ---------- */
let kG = null;   // { nodes, edges, idx, raf, scale, ox, oy, drag, pan, hover, settled }

function kGraphData(){
  const known = {};
  KNOW.forEach(n => { known[n.id] = n; });
  const nodes = KNOW.map(n => ({
    id: n.id, label: n.term || n.id, layer: n.layer || 0,
    verified: n.confidence === 'verified',
    r: (n.layer === 0 ? 22 : 14), x: 0, y: 0, vx: 0, vy: 0
  }));
  const edges = [];
  KNOW.forEach(n => (n.related || []).forEach(r => {
    if (r && known[r.id]) edges.push({ a: n.id, b: r.id });   // 跳过悬空引用
  }));
  const idx = {};
  nodes.forEach(n => { idx[n.id] = n; });
  return { nodes, edges, idx };
}

function kSeed(g, W, H){
  const n = g.nodes.length || 1;
  g.nodes.forEach((nd, i) => {
    const ang = (i / n) * Math.PI * 2;
    const rad = (nd.layer === 0 ? 0.22 : 0.38) * Math.min(W, H);
    nd.x = W/2 + Math.cos(ang)*rad + (Math.random()-0.5)*16;
    nd.y = H/2 + Math.sin(ang)*rad + (Math.random()-0.5)*16;
    nd.vx = nd.vy = 0;
  });
}

function kStep(g, W, H){
  const N = g.nodes;
  for (let i = 0; i < N.length; i++){
    for (let j = i+1; j < N.length; j++){
      const a = N[i], b = N[j];
      let dx = a.x-b.x, dy = a.y-b.y, d2 = dx*dx+dy*dy;
      if (d2 < 1){ dx = Math.random()-0.5; dy = Math.random()-0.5; d2 = 1; }
      const d = Math.sqrt(d2), gap = a.r + b.r + 26;
      const f = (gap*gap)/d2 * 0.85;
      const fx = dx/d*f, fy = dy/d*f;
      a.vx += fx; a.vy += fy; b.vx -= fx; b.vy -= fy;
    }
  }
  const L = 96;
  g.edges.forEach(e => {
    const a = g.idx[e.a], b = g.idx[e.b];
    if (!a || !b) return;
    const dx = b.x-a.x, dy = b.y-a.y;
    const d = Math.max(Math.sqrt(dx*dx+dy*dy), 1);
    const f = (d - L) * 0.012;
    const fx = dx/d*f, fy = dy/d*f;
    a.vx += fx; a.vy += fy; b.vx -= fx; b.vy -= fy;
  });
  let energy = 0;
  N.forEach(nd => {
    nd.vx += (W/2 - nd.x) * 0.0022;
    nd.vy += (H/2 - nd.y) * 0.0022;
    nd.vx *= 0.86; nd.vy *= 0.86;
    nd.x = Math.max(nd.r+6, Math.min(W-nd.r-6, nd.x + nd.vx));
    nd.y = Math.max(nd.r+6, Math.min(H-nd.r-6, nd.y + nd.vy));
    energy += nd.vx*nd.vx + nd.vy*nd.vy;
  });
  return energy;
}

function kDraw(){
  const c = document.getElementById('kgCanvas');
  if (!c || !kG) return;
  const dpr = window.devicePixelRatio || 1;
  const W = c.clientWidth || 700, H = c.clientHeight || 420;
  if (c.width !== Math.round(W*dpr)) { c.width = Math.round(W*dpr); c.height = Math.round(H*dpr); }
  const ctx = c.getContext('2d');
  ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
  ctx.clearRect(0, 0, W, H);
  ctx.save();
  ctx.translate(kG.ox, kG.oy);
  ctx.scale(kG.scale, kG.scale);

  ctx.lineWidth = 1;
  kG.edges.forEach(e => {
    const a = kG.idx[e.a], b = kG.idx[e.b];
    if (!a || !b) return;
    const hot = (kG.hover === e.a || kG.hover === e.b);
    ctx.strokeStyle = hot ? 'rgba(37,99,235,.55)' : 'rgba(148,163,184,.42)';
    ctx.beginPath(); ctx.moveTo(a.x, a.y); ctx.lineTo(b.x, b.y); ctx.stroke();
  });

  const curId = kStack[kStack.length-1];
  kG.nodes.forEach(nd => {
    ctx.beginPath(); ctx.arc(nd.x, nd.y, nd.r, 0, Math.PI*2);
    ctx.fillStyle = nd.layer === 0 ? '#2563eb' : '#94a3b8';
    ctx.globalAlpha = (kG.hover && kG.hover !== nd.id) ? 0.75 : 1;
    ctx.fill();
    ctx.globalAlpha = 1;
    ctx.lineWidth = 2.5;
    ctx.strokeStyle = nd.verified ? '#16a34a' : '#d97706';
    ctx.stroke();
    if (nd.id === curId){
      ctx.beginPath(); ctx.arc(nd.x, nd.y, nd.r+6, 0, Math.PI*2);
      ctx.strokeStyle = '#f59e0b'; ctx.lineWidth = 2; ctx.stroke();
    }
    ctx.font = (nd.layer === 0 ? 'bold 11px ' : '10px ') + '-apple-system,"Microsoft YaHei",sans-serif';
    ctx.fillStyle = nd.layer === 0 ? '#1a2233' : '#6b7488';
    ctx.textAlign = 'center'; ctx.textBaseline = 'middle';
    const lb = nd.label.length > 10 ? nd.label.slice(0,10)+'…' : nd.label;
    ctx.fillText(lb, nd.x, nd.y + nd.r + 10);
  });
  ctx.restore();
}

function kRun(){
  const c = document.getElementById('kgCanvas');
  if (!c || !kG) return;
  const W = c.clientWidth || 700, H = c.clientHeight || 420;
  let e = 1;
  for (let i = 0; i < 12 && e > 0.35; i++) e = kStep(kG, W, H);
  kDraw();
  kG.settled = (e <= 0.35);
  if (kG.settled){ kG.raf = null; return; }          // 收敛后停 rAF，省 CPU
  kG.raf = requestAnimationFrame(kRun);
}

function kWake(){ if (kG && !kG.raf) kG.raf = requestAnimationFrame(kRun); }

function initGraph(){
  const c = document.getElementById('kgCanvas');
  if (!c) { kG = null; return; }
  const W = c.clientWidth || 700, H = c.clientHeight || 420;
  const data = kGraphData();
  kG = { nodes: data.nodes, edges: data.edges, idx: data.idx,
         raf: null, scale: 1, ox: 0, oy: 0, drag: null, pan: null, hover: null, settled: false };
  kSeed(kG, W, H);
  kRun();
}
```

- [ ] **Step 1: 在 `initGraph()` 末尾挂交互**

```js
  const toWorld = (ev) => {
    const r = c.getBoundingClientRect();
    return { x: (ev.clientX - r.left - kG.ox) / kG.scale,
             y: (ev.clientY - r.top  - kG.oy) / kG.scale };
  };
  const hit = (p) => kG.nodes.find(nd => Math.hypot(nd.x-p.x, nd.y-p.y) <= nd.r + 4);

  c.addEventListener('mousedown', ev => {
    const p = toWorld(ev), nd = hit(p);
    if (nd) kG.drag = { nd, dx: p.x-nd.x, dy: p.y-nd.y };
    else kG.pan = { x: ev.clientX - kG.ox, y: ev.clientY - kG.oy };
    c.classList.add('dragging');
  });
  window.addEventListener('mousemove', ev => {
    if (!kG) return;
    if (kG.drag){ const p = toWorld(ev); kG.drag.nd.x = p.x - kG.drag.dx; kG.drag.nd.y = p.y - kG.drag.dy;
                  kG.drag.nd.vx = kG.drag.nd.vy = 0; kDraw(); return; }
    if (kG.pan){ kG.ox = ev.clientX - kG.pan.x; kG.oy = ev.clientY - kG.pan.y; kDraw(); return; }
    const p = toWorld(ev), nd = hit(p);
    const h = nd ? nd.id : null;
    if (h !== kG.hover){ kG.hover = h; kDraw(); }
  });
  window.addEventListener('mouseup', () => {
    if (!kG) return;
    const moved = kG.drag || kG.pan;
    kG.drag = null; kG.pan = null;
    c.classList.remove('dragging');
    if (moved) kWake();                       // 松手后让布局重新收敛
  });
  c.addEventListener('wheel', ev => {
    ev.preventDefault();
    const r = c.getBoundingClientRect();
    const mx = ev.clientX - r.left, my = ev.clientY - r.top;
    const f = ev.deltaY > 0 ? 0.9 : 1.1;
    const ns = Math.min(2.5, Math.max(0.4, kG.scale * f));
    kG.ox = mx - (mx - kG.ox) * (ns / kG.scale);
    kG.oy = my - (my - kG.oy) * (ns / kG.scale);
    kG.scale = ns;
    kDraw();
  }, { passive: false });
  c.addEventListener('click', ev => {
    const p = toWorld(ev), nd = hit(p);
    if (!nd) return;
    kView = 'card';
    kStack = [nd.id];                          // 跳回卡片：直接定位到该节点
    root.innerHTML = knowInner();
    bindKnow();
    flash('已跳回「' + nd.label + '」的卡片');
  });
```
